select distinct category.name
from movie
join film_actor on movie.film_id = film_actor.film_id
join film_category on film_category.film_id = movie.film_id
join category on category.category_id = film_category.category_id
where film_actor.actor_id = 148;

select es.last_name as employee_last_name, ms.last_name as manager_last_name
from staff as es
left join staff as ms on ms.employee_id = es.reports_to;

SELECT name, invoice_date::date
FROM track
left join invoice_line on invoice_line.track_id = track.track_id
left join invoice on invoice.invoice_id = invoice_line.invoice_id;

select movie.title from movie
left join film_actor on movie.film_id = film_actor.film_id where film_actor.film_id is null;

select artist.name
from artist
left join album on artist.artist_id = album.artist_id
where album.artist_id is null;

select count(*) from movie m
join film_category f on f.film_id = m.film_id
join category c on c.category_id = f.category_id
where c.name ~* '^c';

select count(distinct t.album_id) from track t
join genre g on g.genre_id = t.genre_id
where g.name ~* 'jazz';

SELECT
    sum(total),
    CASE
        WHEN i.total < 1 then 'low cost'
        ELSE 'high cost'
    END cost_category
FROM invoice i
join client c on c.customer_id = i.customer_id
where c.phone is not null
group by cost_category;

SELECT
    c.name,
    count(distinct fa.actor_id) actor_count
FROM film_actor fa
join film_category fc on fc.film_id = fa.film_id
join category c on c.category_id = fc.category_id
group by c.name
order by actor_count desc, c.name asc;

SELECT m.rating, sum(rental_rate)/count(*)
FROM movie m
GROUP BY m.rating
HAVING avg(m.rental_rate) > 3;

SELECT i.invoice_date::date, sum(i.total)
FROM invoice i
WHERE i.invoice_date between '2011-09-01' and '2011-10-01'
GROUP BY i.invoice_date
HAVING min(i.total) > 1 and max(i.total) < 10;

select i.billing_country, count(*)
from invoice i
WHERE i.billing_address !~* '(Street|Way|Road|Drive)' and i.billing_postal_code is null
GROUP by i.billing_country
having count(*) > 10;

select a.title, sum(t.unit_price) as album_price
from album a
join track t on t.album_id = a.album_id
group by a.title
having sum(t.unit_price) < 5
order by sum(t.unit_price) desc;

select c.first_name, count(c.first_name)
from client c
group by c.first_name
having count(c.first_name) > 1;

SELECT customer_id,
       invoice_date,
       total,
       DATE_TRUNC('month', CAST(invoice_date AS timestamp)),
       EXTRACT(WEEK FROM CAST(invoice_date AS timestamp))
FROM invoice
WHERE
(customer_id BETWEEN 20 AND 50)
and (
    EXTRACT(WEEK FROM CAST(invoice_date AS timestamp)) = 5
    or EXTRACT(WEEK FROM CAST(invoice_date AS timestamp)) = 7
    or EXTRACT(WEEK FROM CAST(invoice_date AS timestamp)) = 10
    or EXTRACT(WEEK FROM CAST(invoice_date AS timestamp)) = 33
    or EXTRACT(WEEK FROM CAST(invoice_date AS timestamp)) = 48
);

select i.*
from invoice i
join client c on c.customer_id = i.customer_id
where EXTRACT(DAY FROM i.invoice_date::timestamp) = 1 and c.company is null;

select c.name, STRING_AGG(m.title, ', '  ORDER BY m.title) as titles
from movie m
join film_category fc on fc.film_id = m.film_id
join category c on c.category_id = fc.category_id
group by c.name;
