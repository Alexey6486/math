create table trains (id serial primary key, train_number varchar(13), departure_station integer references it_stations(id), departure_date date, arrival_station integer references it_stations(id), arrival_date date);
CREATE TABLE trains (
    id SERIAL PRIMARY KEY ,
    train_number varchar(13),
    departure_station integer REFERENCES stations,
    departure_date timestamp,
    arrival_station integer REFERENCES stations,
    arrival_date timestamp
);

create table car_types (id integer primary key, car_type_name text);
CREATE TABLE car_types (
    id integer PRIMARY KEY,
    car_type_name text
);

create table cars (car_number integer primary key, car_type_id integer references car_types default 1, cars_have_no_passports boolean, date_entry date);
CREATE TABLE cars (
    id SERIAL PRIMARY KEY,
    car_number varchar(6),
    car_type_id integer REFERENCES car_types
);

create table cars_in_trains (id serial primary key, car_id integer references cars(car_number), train_id integer references trains(id));
alter table cars_in_trains add column attach_date date, add column detach_date date;
CREATE TABLE cars_in_trains (
    id SERIAL PRIMARY KEY,
    train_id integer REFERENCES trains,
    car_id integer REFERENCES cars,
    attach_date timestamp,
    detach_date timestamp
);

create table cars_without_passport (id serial primary key, car_id integer references cars(car_number) unique);
CREATE TABLE cars_have_no_passports (
    id SERIAL PRIMARY KEY,
    car_id integer REFERENCES cars UNIQUE,
    date_entry timestamp
);
